# 最新B表匯出 SQL（DEPT='MPW1'）：風險評估＋高風險＋機會

## 0. 摘要（可直接當 PROMPT 用）

```
目的：產出部門 MPW1 最新版 B 表 15 欄＋REFNO，供匯出 CSV。
輸出欄：ITEMNO(項次,ROW_NUMBER)、KEYNO(風險評估編號,dept-flow0-flow1-srlNo)、
FLOWNAME(作業流程/名稱,RE01)、WORKSTUS(狀態,RE02解碼0=正常/1=保養/2=異常處理)、
PROCEDURE(步驟)、REASON(可能發生原因)、EFFECT(後果影響)、ACCIDENTTYPE(事故類型)、
CONSTRUCTCTRL(工程控制)、MANAGECTRL(管理控制)、EQUIPMENTCTRL(個人防護具)、
EXPOSE(暴露頻率)、FREQUENCE(發生機率)、SEVERITY(嚴重度)、RANK(風險等級)、
REFNO(關聯文件編號,RA-/HR-/HC-/RO-+文號)。
規則：
1) 不分 RECode 第4碼(Y/S)，取 DEPT='MPW1' 且 tbeaworkdoc.DOCSTATUS='E'
   且 REPORTID='0002'(風險與機會評估維護階段)，
   且 INNER JOIN tbeaAnnotate（無簽核紀錄排除），
   以 MAX(tbeaAnnotate.TIMESTAMP) 最新者為最新版風險評估(LATEST_RA)。
2) B 表原型對齊 eajcRERpt.java exportType=B：RE03左接RE00/RE01(parentCode/flowCode)/RE02；
   RE02 不對 SRLNO，改 LOCATE(TRIM(RE03.ACCIDENTTYPE)||';',RE02.DAMAGE)>0，
   一對多時按 RE02.SRLNO 最小只取首筆；WORKSTUS 沿原碼解碼。
3) RANK IN(1,2) 才看高風險：高風險文號經 WorkDocEX
   (FIELD1~5=FORMNO+FLOW0+FLOW1+SRLNO+VERSDATE→MAX(WORKDOCID)，同鍵多文號取最大)
   取每鍵最新 VERSDATE 版(RE04)，
   前3欄改善取 RE04，後4欄(EXPOSE/FREQUENCE/SEVERITY/RANK)取 RE042 同 VERSDATE 下
   最新 expWDate 且 STATUSCODE='30'(完工評估,RANK為計算值)；須 RE04 簽核 E 才覆蓋(HR)，
   否則有高風險資料僅改 REFNO(HC)，無則 RA。
4) RANK 非1/2 且 TBEARE13.IMETHOD IN('D','I')(D=消除/I=取代)才用 RE13 後7欄覆蓋(RO)。
5) REFNO(RA-/HR-/HC-/RO-+文號)：RA-=風險評估文號、HR-=高風險文號、HC-=高風險未完成文號、RO-=機會(接風險文號)。
限制：無 TBEARE13A，一律 TBEARE13；DB2 語法(||/LOCATE/CHAR/FETCH FIRST 1 ROW ONLY)。
```

## 1. 完整 SQL

```sql
WITH LATEST_RA AS (
  /* 最新版風險評估文號：MPW1 + 簽核完成 + Annotate時間戳記最新 */
  SELECT A.FORMNO
  FROM DB.TBEARE00 A
  JOIN DB.TBEAWORKDOC W
    ON W.WORKDOCID = A.FORMNO
   AND W.DOCSTATUS = 'E'
   AND W.REPORTID = '0002'   /* 0002=風險與機會評估維護階段 */
  JOIN DB.TBEAANNOTATE N
    ON N.WORKDOCID = W.WORKDOCID
  WHERE A.DEPT = 'MPW1'
  GROUP BY A.FORMNO
  ORDER BY MAX(N.TIMESTAMP) DESC, MAX(A.FORMNO) DESC
  FETCH FIRST 1 ROW ONLY
),
BASE_RAW AS (
  /* B表原型（eajcRERpt.java exportType=B）；D可能一對多，先保留再收斂 */
  SELECT
    A.COMPID,
    A.FORMNO,
    B.DEPT || '-' || A.FLOW0CODE || '-' || A.FLOW1CODE || '-' || CHAR(A.SRLNO)
      AS KEYNO,
    C.FLOWNAME,
    D.WORKSTUS AS D_WS,
    D.SRLNO    AS D_SRLNO,
    A.PROCEDURE, A.REASON, A.EFFECT, A.ACCIDENTTYPE,
    A.CONSTRUCTCTRL, A.MANAGECTRL, A.EQUIPMENTCTRL,
    A.EXPOSE, A.FREQUENCE, A.SEVERITY, A.RANK,
    A.FLOW0CODE, A.FLOW1CODE, A.SRLNO
  FROM DB.TBEARE03 A
  JOIN LATEST_RA L ON L.FORMNO = A.FORMNO
  LEFT JOIN DB.TBEARE00 B
    ON A.COMPID=B.COMPID AND A.FORMNO=B.FORMNO
  LEFT JOIN DB.TBEARE01 C
    ON A.COMPID=C.COMPID AND A.FORMNO=C.FORMNO
   AND A.FLOW0CODE=C.PARENTCODE AND A.FLOW1CODE=C.FLOWCODE
  LEFT JOIN DB.TBEARE02 D
    ON A.COMPID=D.COMPID AND A.FORMNO=D.FORMNO
   AND A.FLOW0CODE=D.FLOW0CODE AND A.FLOW1CODE=D.FLOW1CODE
   /* SRLNO不對接：以A.ACCIDENTTYPE||';'存在於D.DAMAGE之中關聯(DB2:LOCATE) */
   AND TRIM(COALESCE(A.ACCIDENTTYPE,'')) <> ''
   AND LOCATE(TRIM(A.ACCIDENTTYPE) || ';', D.DAMAGE) > 0
),
BASE AS (
  /* 同ON條件下僅取第一筆D（按D.SRLNO最小；要改正常優先請改ORDER BY） */
  SELECT
    COMPID, FORMNO, KEYNO, FLOWNAME,
    CASE WHEN D_WS='0' THEN '正常'
         WHEN D_WS='1' THEN '保養'
         WHEN D_WS='2' THEN '異常處理'
         ELSE '' END AS WORKSTUS,
    PROCEDURE, REASON, EFFECT, ACCIDENTTYPE,
    CONSTRUCTCTRL, MANAGECTRL, EQUIPMENTCTRL,
    EXPOSE, FREQUENCE, SEVERITY, RANK,
    FLOW0CODE, FLOW1CODE, SRLNO
  FROM (
    SELECT R.*,
      ROW_NUMBER() OVER (
        PARTITION BY COMPID,FORMNO,FLOW0CODE,FLOW1CODE,SRLNO
        ORDER BY D_SRLNO
      ) AS RN
    FROM BASE_RAW R
  ) X WHERE X.RN = 1
),
/* 高風險：改善措施取RE04最新版，完工評估4欄取RE042(狀態30) */
HR_LATEST AS (
  SELECT X.* FROM (
    SELECT
      H.COMPID, H.FORMNO, H.FLOW0CODE, H.FLOW1CODE, H.SRLNO, H.VERSDATE,
      H.CONSTRUCTCTRL, H.MANAGECTRL, H.EQUIPMENTCTRL,
      EX.WORKDOCID AS HRDOCNO,
      HW.DOCSTATUS  AS HRDOCSTATUS,
      ROW_NUMBER() OVER (
        PARTITION BY H.COMPID,H.FORMNO,H.FLOW0CODE,H.FLOW1CODE,H.SRLNO
        ORDER BY H.VERSDATE DESC, EX.WORKDOCID DESC
      ) AS SN
    FROM DB.TBEARE04 H
    JOIN LATEST_RA L ON L.FORMNO = H.FORMNO
    /* 同FIELD1~5取WORKDOCID最大者(圖:同VERSDATE多文號取最大) */
    LEFT JOIN (SELECT FIELD1,FIELD2,FIELD3,FIELD4,FIELD5,
                      MAX(WORKDOCID) AS WORKDOCID
               FROM DB.TBEAWORKDOCEX
               GROUP BY FIELD1,FIELD2,FIELD3,FIELD4,FIELD5) EX
      ON EX.FIELD1 = H.FORMNO
     AND EX.FIELD2 = H.FLOW0CODE
     AND EX.FIELD3 = H.FLOW1CODE
     AND EX.FIELD4 = H.SRLNO
     AND EX.FIELD5 = H.VERSDATE
    LEFT JOIN DB.TBEAWORKDOC HW
      ON HW.WORKDOCID = EX.WORKDOCID
  ) X WHERE X.SN = 1
),
/* 完工評估(RE042)：同VERSDATE下取最新expWDate且狀態=30完工評估 */
HR_EVAL AS (
  SELECT X.* FROM (
    SELECT
      E.COMPID, E.FORMNO, E.FLOW0CODE, E.FLOW1CODE, E.SRLNO, E.VERSDATE,
      E.EXPOSE, E.FREQUENCE, E.SEVERITY, E.RANK, E.STATUSCODE,
      E.EXPWDATE, E.ACTWDATE, E.EXPEDATE, E.ACTEDATE,
      ROW_NUMBER() OVER (
        PARTITION BY E.COMPID,E.FORMNO,E.FLOW0CODE,E.FLOW1CODE,E.SRLNO,E.VERSDATE
        ORDER BY E.EXPWDATE DESC, E.ACTEDATE DESC
      ) AS SN
    FROM DB.TBEARE042 E
    JOIN HR_LATEST H
      ON H.COMPID=E.COMPID AND H.FORMNO=E.FORMNO
     AND H.FLOW0CODE=E.FLOW0CODE AND H.FLOW1CODE=E.FLOW1CODE
     AND H.SRLNO=E.SRLNO AND H.VERSDATE=E.VERSDATE
    WHERE E.STATUSCODE = '30'
  ) X WHERE X.SN = 1
),
HR_DONE AS (
  /* 已完工且簽核完成：RE04簽核E + RE042有30 */
  SELECT H.*, E.EXPOSE AS E_EXPOSE, E.FREQUENCE AS E_FREQUENCE,
         E.SEVERITY AS E_SEVERITY, E.RANK AS E_RANK
  FROM HR_LATEST H
  JOIN HR_EVAL E
    ON E.COMPID=H.COMPID AND E.FORMNO=H.FORMNO
   AND E.FLOW0CODE=H.FLOW0CODE AND E.FLOW1CODE=H.FLOW1CODE
   AND E.SRLNO=H.SRLNO AND E.VERSDATE=H.VERSDATE
  WHERE H.HRDOCSTATUS = 'E'
),
HR_ANY AS (
  SELECT * FROM HR_LATEST
),
RO AS (
  /* 機會 D/I（exptByFormNox接法） */
  SELECT X.* FROM (
    SELECT O.*,
      ROW_NUMBER() OVER (
        PARTITION BY O.COMPID,O.FORMNO,O.FLOW0CODE,O.FLOW1CODE,O.SRLNO
        ORDER BY O.FORMNO DESC
      ) AS SN
    FROM DB.TBEARE13 O
    JOIN LATEST_RA L ON L.FORMNO = O.FORMNO
    WHERE O.IMETHOD IN ('D','I')
  ) X WHERE X.SN = 1
)
SELECT
  ROW_NUMBER() OVER (ORDER BY B.FLOW0CODE,B.FLOW1CODE,B.SRLNO) AS ITEMNO,
  B.KEYNO, B.FLOWNAME, B.WORKSTUS,
  B.PROCEDURE, B.REASON, B.EFFECT, B.ACCIDENTTYPE,
  COALESCE(HD.CONSTRUCTCTRL,
           CASE WHEN B.RANK NOT IN ('1','2') THEN O.CONSTRUCTCTRL END,
           B.CONSTRUCTCTRL) AS CONSTRUCTCTRL,
  COALESCE(HD.MANAGECTRL,
           CASE WHEN B.RANK NOT IN ('1','2') THEN O.MANAGECTRL END,
           B.MANAGECTRL)    AS MANAGECTRL,
  COALESCE(HD.EQUIPMENTCTRL,
           CASE WHEN B.RANK NOT IN ('1','2') THEN O.EQUIPMENTCTRL END,
           B.EQUIPMENTCTRL) AS EQUIPMENTCTRL,
  COALESCE(HD.E_EXPOSE,
           CASE WHEN B.RANK NOT IN ('1','2') THEN O.EXPOSE END,
           B.EXPOSE)        AS EXPOSE,
  COALESCE(HD.E_FREQUENCE,
           CASE WHEN B.RANK NOT IN ('1','2') THEN O.FREQUENCE END,
           B.FREQUENCE)     AS FREQUENCE,
  COALESCE(HD.E_SEVERITY,
           CASE WHEN B.RANK NOT IN ('1','2') THEN O.SEVERITY END,
           B.SEVERITY)      AS SEVERITY,
  COALESCE(HD.E_RANK,
           CASE WHEN B.RANK NOT IN ('1','2') THEN O.RANK END,
           B.RANK)          AS RANK,
  CASE
    WHEN B.RANK IN ('1','2') AND HD.HRDOCNO IS NOT NULL
      THEN 'HR-' || TRIM(HD.HRDOCNO)
    WHEN B.RANK IN ('1','2') AND HA.HRDOCNO IS NOT NULL
      THEN 'HC-' || TRIM(HA.HRDOCNO)
    WHEN B.RANK NOT IN ('1','2') AND O.FORMNO IS NOT NULL
      THEN 'RO-' || TRIM(B.FORMNO)
    ELSE 'RA-' || TRIM(B.FORMNO)
  END AS REFNO
FROM BASE B
LEFT JOIN HR_DONE HD
  ON HD.COMPID=B.COMPID AND HD.FORMNO=B.FORMNO
 AND HD.FLOW0CODE=B.FLOW0CODE AND HD.FLOW1CODE=B.FLOW1CODE
 AND HD.SRLNO=B.SRLNO
 AND B.RANK IN ('1','2')
LEFT JOIN HR_ANY HA
  ON HA.COMPID=B.COMPID AND HA.FORMNO=B.FORMNO
 AND HA.FLOW0CODE=B.FLOW0CODE AND HA.FLOW1CODE=B.FLOW1CODE
 AND HA.SRLNO=B.SRLNO
 AND B.RANK IN ('1','2')
LEFT JOIN RO O
  ON O.COMPID=B.COMPID AND O.FORMNO=B.FORMNO
 AND O.FLOW0CODE=B.FLOW0CODE AND O.FLOW1CODE=B.FLOW1CODE
 AND B.SRLNO = INTEGER(TRIM(O.SRLNO))
 AND B.RANK NOT IN ('1','2')
ORDER BY B.FLOW0CODE, B.FLOW1CODE, B.SRLNO;
```

## 2. 詳細說明

### 2.1 資料來源與版本依據
- 程式依據：`src/com/chsteel/ea/bp/eajcRERpt.java`
  - B 表（`exportType=B` L46-59）：`RE03(a)` 左接 `RE00(b)`、`RE01(c:parentCode/flowCode)`、`RE02(d)`；
    `KEYNO=dept-flow0-flow1-char(srlNo)`；`WORKSTUS` 解碼 `0=正常/1=保養/2=異常處理/ELSE ''`（L50，`eajjRE02MainN.jsp:91-93` 同）。
  - 機會（`exptByFormNox` L98-113）：左接 `RE13(e)`，鍵 `a.srlNo=int(e.srlNo)`，`imethod D=消除/I=取代`。
  - 高風險（`exportHighRisk` L149-150）：`RE04` 按每鍵 `VERSDATE DESC` 取 `SN=1`，且只看 `RANK IN(1,2)`。
- 高風險文號依據：`eajcWorkDocEX_dao.queryWorkDocEX`（L491-499）＋圖確認：
  `FIELD1~5=FORMNO+FLOW0+FLOW1+SRLNO(數字)+VERSDATE→MAX(WORKDOCID)`（同鍵多文號取最大）。
- 無 `TBEARE13A`（虛擬表），`FLOWNAME` 與機會改善一律用 `TBEARE13`。
- 最新版定義：`RE00.DEPT='MPW1'`＋`WORKDOC(WORKDOCID=FORMNO).DOCSTATUS='E'`＋`REPORTID='0002'`（風險與機會評估維護階段），
  以 `MAX(ANNOTATE.TIMESTAMP)` 最大者為最新（`INNER JOIN`，無簽核紀錄直接排除）；
  不分 `RECode` 第4碼 Y/S；`ANNOTATETYPE` 若只計主管請加掛。

### 2.2 CTE 逐段說明
| CTE | 輸入→輸出 | 關鍵邏輯 |
|---|---|---|
| `LATEST_RA` | `RE00+WORKDOC+ANNOTATE`→單一 `FORMNO` | `GROUP BY FORMNO ORDER BY MAX(TIMESTAMP) DESC FETCH FIRST 1 ROW ONLY` |
| `BASE_RAW` | `RE03+RE00/RE01/RE02`→B 表原型（含重複） | `RE02` 不對 `SRLNO`，改 `LOCATE(TRIM(ACCIDENTTYPE)||';',DAMAGE)>0`（加分號避 `PH1→PH10` 誤匹）；保留 `D_WS/D_SRLNO` 供收斂 |
| `BASE` | `BASE_RAW`→每 RE03 一列 | `ROW_NUMBER() PARTITION BY 5鍵 ORDER BY D_SRLNO` 取 `RN=1`，再解碼 `WORKSTUS` |
| `HR_LATEST` | `RE04+WORKDOCEX+WORKDOC`→每鍵最新改善版 | `PARTITION BY 5鍵 ORDER BY VERSDATE DESC,WORKDOCID DESC SN=1`；`HRDOCNO/HRDOCSTATUS` 供 HR/HC 判斷 |
| `HR_EVAL` | `RE042`→完工評估值 | 同 `VERSDATE` 下 `STATUSCODE='30'`，按 `expWDate DESC,actEDate DESC SN=1`；`RANK` 為計算值（對應畫面紅框） |
| `HR_DONE` | `HR_LATEST⋈HR_EVAL`＋簽核 | 條件 `HRDOCSTATUS='E'`；輸出 `E_EXPOSE/E_FREQUENCE/E_SEVERITY/E_RANK` |
| `HR_ANY` | `HR_LATEST` 全集 | 供 `HC`（有高風險但未完工／未簽核）判斷 |
| `RO` | `RE13(IMETHOD D/I)`→每鍵一筆 | `PARTITION BY 5鍵`，`O.SRLNO` 字串時 `INTEGER(TRIM())` 對接 |
| 終查 | `BASE/HD/HA/O`→16 欄 | 前8欄取 `BASE`；後7欄優先序 `HD>RO>BASE`（`HD` 中前3取 `RE04`、後4取 `RE042`）；`REFNO` 按 `HR/HC/RO/RA`；`ITEMNO=ROW_NUMBER()` |

### 2.3 REFNO 與覆蓋矩陣
| BASE.RANK | 高風險 | 機會 | 後7欄 | REFNO |
|---|---|---|---|---|
| 1/2 | 有 `HD`（簽核E＋RE042有30） | — | 前3=`RE04`，後4=`RE042` | `HR-+HRDOCNO` |
| 1/2 | 僅 `HA`（未完工或未簽核） | — | 維持 `BASE` | `HC-+HRDOCNO` |
| 非1/2 | — | 有 `RO(D/I)` | 7欄=`RE13` | `RO-+FORMNO` |
| 其他 | — | — | 維持 `BASE` | `RA-+FORMNO` |

### 2.4 已知取捨與上線前必換
- `BASE→RE02` 一對多已收斂為首筆（`D.SRLNO` 最小；要正常優先改 `ORDER BY CASE WHEN D_WS='0'...`）；`ROW_NUMBER` 版較關聯子查詢快（`LOCATE` 只算一次）。
- `HRDOCNO` 為 NULL＝該改善版尚未立案（無 WORKDOCEX），`HC` 不觸發、維持 `RA`。
- 表名：`DB.TBEARExx/TBEAWORKDOC(EX)/TBEAANNOTATE` ↔ `db.tbearexx/db.tbeaWorkDoc(EX)/db.tbeaAnnotate`。
- 保留字：`N.TIMESTAMP`、`EX.FIELDx` 必要時加雙引號。
- `O.SRLNO` 已為 INT 時，`INTEGER(TRIM(O.SRLNO))` 改 `O.SRLNO`。
- DB2→SQL Server：`||`→`+`、`CHAR(x)`→`CAST`、`LOCATE(a,b)`→`CHARINDEX(a,b)`。
